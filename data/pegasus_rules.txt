#   md5:          <hex>                HARD - APK MD5
#   cert:         <substring>          HARD - signer DN substring (device scans)
#   cert_sha256:  <hex>                HARD - signer certificate SHA-256
#   lib:          <libname.so>         HARD - native library file inside the APK
#   asset:        <relative path>      HARD - asset path inside the APK
#   perm:         <permission>         SOFT - must be present twice per block
#   string:       <substring>          SOFT - DEX string (twice per block)
#   class:        <substring>          SOFT - manifest component (twice per block)
#   domain:       <substring>          searched in DEX strings
#   ip:           <substring>          searched in DEX strings
#   url:          <substring>          searched in DEX strings
#   email:        <substring>          searched in DEX strings
